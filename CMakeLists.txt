cmake_minimum_required(VERSION 3.24)
project(GameInputProbe VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(GAMEINPUT_BUILD_PROBE "Build Windows x64 Detours DLL" ${WIN32})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(GLOB_RECURSE identity_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.cmake")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${identity_sources})
set(identity_input "${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${CMAKE_CXX_FLAGS};${CMAKE_CXX_FLAGS_RELEASE};${CMAKE_CXX_FLAGS_DEBUG};${CMAKE_SIZEOF_VOID_P}")
foreach(source IN LISTS identity_sources)
    file(SHA256 "${source}" source_hash)
    string(APPEND identity_input ";${source_hash}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" cmake_hash)
string(SHA256 build_source_hash "${identity_input};${cmake_hash}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/$<CONFIG>/build_identity.hpp"
    CONTENT "#pragma once\n#define GAMEINPUT_BUILD_ID \"${build_source_hash};$<CONFIG>\"\n")
include_directories("${CMAKE_BINARY_DIR}/generated/$<CONFIG>")
if(BUILD_TESTING)
    add_executable(throttle_policy_tests tests/native/policy_tests.cpp)
    target_include_directories(throttle_policy_tests PRIVATE src)
    add_test(NAME throttle_policy COMMAND throttle_policy_tests)
endif()
if(GAMEINPUT_BUILD_PROBE)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The probe requires MSVC Windows x64; portable tests: -DGAMEINPUT_BUILD_PROBE=OFF")
    endif()
    include(cmake/Detours.cmake)
    add_library(probe_engine STATIC src/probe/engine.cpp)
    target_include_directories(probe_engine PUBLIC src)
    target_compile_definitions(probe_engine PUBLIC _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(probe_engine PRIVATE /W4 /permissive- /EHsc /guard:cf)
    add_library(GameInputProbe SHARED src/probe/probe.cpp src/probe/identity.cpp src/probe/mapping.cpp)
    target_link_libraries(GameInputProbe PRIVATE probe_engine detours advapi32 bcrypt version shell32 ole32)
    target_compile_options(GameInputProbe PRIVATE /W4 /permissive- /EHsc /guard:cf)
    target_link_options(GameInputProbe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    add_custom_command(TARGET GameInputProbe POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_BINARY_DIR}/licenses/Detours-LICENSE.md" "$<TARGET_FILE_DIR:GameInputProbe>/Detours-LICENSE.md")
    install(TARGETS GameInputProbe RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(probe_integration_tests tests/native/integration_tests.cpp src/probe/mapping.cpp)
        target_link_libraries(probe_integration_tests PRIVATE probe_engine detours advapi32)
        add_test(NAME probe_integration COMMAND probe_integration_tests $<TARGET_FILE:GameInputProbe>)
        set_tests_properties(probe_integration PROPERTIES TIMEOUT 30)
        add_executable(remote_image_tests tests/native/remote_image_tests.cpp)
        target_include_directories(remote_image_tests PRIVATE src)
        add_test(NAME remote_image COMMAND remote_image_tests $<TARGET_FILE:GameInputProbe>)
        add_executable(transaction_tests tests/native/transaction_tests.cpp)
        target_link_libraries(transaction_tests PRIVATE detours)
        add_test(NAME transaction_heap_lock COMMAND transaction_tests)
        set_tests_properties(transaction_heap_lock PROPERTIES TIMEOUT 10)
    endif()
endif()
# Controller is independently owned; automatically include it once supplied.
if(WIN32 AND GAMEINPUT_BUILD_PROBE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/controller/CMakeLists.txt")
    add_subdirectory(src/controller)
elseif(WIN32 AND GAMEINPUT_BUILD_PROBE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/controller/main.cpp")
    add_executable(GameInputProbeController src/controller/main.cpp)
    target_compile_features(GameInputProbeController PRIVATE cxx_std_20)
    target_include_directories(GameInputProbeController PRIVATE src)
    target_link_libraries(GameInputProbeController PRIVATE bcrypt version wintrust crypt32 advapi32 psapi)
endif()
